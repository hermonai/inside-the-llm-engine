cmake_minimum_required(VERSION 3.19)
project(book_gguf_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED GGML_SOURCE)
  message(FATAL_ERROR "Supply -DGGML_SOURCE=/absolute/path/to/pinned/ggml")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GGML_SOURCE}" rev-parse HEAD
                OUTPUT_VARIABLE GGML_REV OUTPUT_STRIP_TRAILING_WHITESPACE
                COMMAND_ERROR_IS_FATAL ANY)
if(NOT GGML_REV STREQUAL "353b63b439f27ab2cc19dac97ab1681ba6d2d084")
  message(FATAL_ERROR "This reference check requires ggml revision 353b63b")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GGML_SOURCE}" diff --quiet HEAD
                RESULT_VARIABLE GGML_DIRTY)
if(NOT GGML_DIRTY EQUAL 0)
  message(FATAL_ERROR "Pinned ggml source has tracked modifications")
endif()
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" "${CMAKE_BINARY_DIR}/ggml")
add_executable(gguf-reference main.cpp)
target_link_libraries(gguf-reference PRIVATE ggml ggml-cpu ggml-base)
