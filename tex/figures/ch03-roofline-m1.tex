% Roofline of an Apple M1 GPU with measured llama.cpp points.

% Roofs: compute 2.6 TFLOP/s (Apple); bandwidth about 68 GB/s nominal (from Apple's statements; 68.3 plotted),
% 49 GB/s achieved by a streaming kernel (measured, median of copy tests).

\begin{tikzpicture}

  legend style={font=\footnotesize, at={(0.98,0.03)}, anchor=south east, draw=black!30, fill=white},
  legend cell align=left,
]

\addlegendentry{compute roof, 2.6 TFLOP/s (Apple)}

\addlegendentry{nominal bandwidth, about 68 GB/s (Apple)}
